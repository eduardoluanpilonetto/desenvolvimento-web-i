05 Salvar no LocalStorage

Crie um formulário com um campo "Nome" e um botão que salve o valor no LocalStorage.

demonstração funcional
<input type="text" id="nome">
<button onclick="salvar()">Salvar</button>

<script>
function salvar() {
  let nome = document.getElementById("nome").value;
  localStorage.setItem("nome", nome);
  alert("Salvo!");
}
</script>

06 Recuperar automaticamente

Ao carregar a página, recupere automaticamente o nome salvo no LocalStorage.

demonstração — recarregue a página após salvar no exercício 5
window.onload = function() {
  let nome = localStorage.getItem("nome");
  if (nome) {
    document.getElementById("nome").value = nome;
  }
}

07 Remover do LocalStorage

Crie um botão para remover o nome armazenado no LocalStorage.

demonstração funcional
<button onclick="remover()">Remover</button>

<script>
function remover() {
  localStorage.removeItem("nome");
  alert("Removido!");
}
</script>

08 Geolocation API

Utilize a Geolocation API para exibir latitude e longitude do usuário na tela.

demonstração funcional (o navegador pedirá permissão)
navigator.geolocation.getCurrentPosition(mostrarPosicao);

function mostrarPosicao(posicao) {
  console.log(posicao.coords.latitude);
  console.log(posicao.coords.longitude);
}

09 Clipboard API

Crie um botão que copie a frase "Aprendendo HTML5" para a área de transferência.

demonstração funcional
<button onclick="copiar()">Copiar</button>

<script>
function copiar() {
  navigator.clipboard.writeText("Aprendendo HTML5");
}
</script>

Código completo:

<body>
  <header><h1>Portal HTML5</h1></header>

  <nav>
    <a href="#">Início</a>
    <a href="#">Mídia</a>
    <a href="#">Sobre</a>
  </nav>

  <main>
    <section>
      <h2>Mídia</h2>
      <audio controls>
        <source src="musica.mp3" type="audio/mpeg">
      </audio>
      <video controls width="400">
        <source src="video.mp4" type="video/mp4">
      </video>
    </section>

    <section>
      <h2>Seus dados</h2>
      <input type="text" id="nome">
      <button onclick="salvar()">Salvar</button>
      <button onclick="remover()">Remover</button>
    </section>

    <section>
      <h2>Localização</h2>
      <button onclick="obterLocalizacao()">Obter localização</button>
    </section>

    <section>
      <h2>Compartilhar</h2>
      <button onclick="copiar()">Copiar frase</button>
    </section>
  </main>

  <footer>© 2025</footer>

  <script>
    window.onload = function() {
      let nome = localStorage.getItem("nome");
      if (nome) document.getElementById("nome").value = nome;
    }

    function salvar() {
      let nome = document.getElementById("nome").value;
      localStorage.setItem("nome", nome);
    }

    function remover() {
      localStorage.removeItem("nome");
    }

    function obterLocalizacao() {
      navigator.geolocation.getCurrentPosition(function(pos) {
        console.log(pos.coords.latitude, pos.coords.longitude);
      });
    }

    function copiar() {
      navigator.clipboard.writeText("Aprendendo HTML5");
    }
  </script>
</body>
